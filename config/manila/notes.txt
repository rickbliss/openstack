# Ensure /etc/ceph exists on each host, then copy ceph.conf and client.manila keyring
for host in localhost bc00 bc01 bc02 bc03 bc04; do
  ssh "$host" "sudo mkdir -p /etc/ceph"
  sudo ceph auth get-or-create client.manila \
    | ssh "$host" sudo tee /etc/ceph/ceph.client.manila.keyring
  sudo scp /etc/ceph/ceph.conf "$host":/etc/ceph/ceph.conf
done

# Create metadata and data pools (adjust PG counts as needed)
sudo ceph osd pool create cephfs_hdd_metadata 128
sudo ceph osd pool create cephfs_hdd_data     128

# Create the CephFS filesystem using those pools
sudo ceph fs volume create cephfs_hdd \
  cephfs_hdd_metadata cephfs_hdd_data

# Configure MDS for high availability
sudo ceph fs volume set cephfs_hdd max_mds 3
sudo ceph fs volume set cephfs_hdd allow_standby_replay true

cat << EOF > nfs-manila.yaml
service_type: nfs
service_id:   manila-ganesha
placement:
  hosts:
    - <YOUR_STORAGE_HOST>    # e.g. bc04
spec:
  port: 2049
EOF

sudo ceph orch apply -i nfs-manila.yaml


sudo ceph nfs export create cephfs \
  manila-ganesha   /cephfs   cephfs_hdd --path=/

sudo mkdir -p /etc/kolla/config/manila
sudo cp /etc/ceph/ceph.conf \
       /etc/kolla/config/manila/ceph.conf
sudo cp /etc/ceph/ceph.client.manila.keyring \
       /etc/kolla/config/manila/ceph.client.manila.keyring
